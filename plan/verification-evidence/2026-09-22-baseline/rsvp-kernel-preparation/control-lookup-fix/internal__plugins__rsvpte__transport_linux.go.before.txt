//go:build linux

// Design: docs/architecture/rsvpte/mpls-rsvp-te.md -- Linux RSVP packet I/O.
// Related: transport.go -- protocol identities and forwarding context.
//
// The wildcard protocol-46 receiver diverts transit Router Alert packets.
// Replies use a separate socket and the outgoing interface address. PATH uses
// IP_HDRINCL: the sockaddr selects the next hop while the header retains the
// sender and SESSION destination. Kernel FIB ownership stays with fib-kernel.
package rsvpte

import (
	"encoding/binary"
	"errors"
	"fmt"
	"io"
	"net"
	"net/netip"
	"os"
	"sync"
	"syscall"

	"github.com/vishvananda/netlink"
	"golang.org/x/sys/unix"
)

// rsvpProtocol is the IANA IP protocol number for RSVP (RFC 2205 Section 3.3).
const rsvpProtocol = 46

// rawTransport is safe for concurrent use. The caller MUST call Close to stop
// its receive worker and release its sockets. sendMu also protects descriptor
// lifetime, so Close cannot recycle a descriptor during a send or route query.
type rawTransport struct {
	sendMu sync.Mutex
	sendFD int
	pathFD int
	routes *netlink.Handle
	rx     *os.File
	rxRaw  syscall.RawConn
	recvCh chan Packet
	closeC chan struct{}
	done   chan struct{}
	closed bool
	once   sync.Once
	closeErr error
}

func openRawTransport(localAddr netip.Addr) (Transport, error) {
	if localAddr.IsValid() {
		if !localAddr.Is4() {
			return nil, fmt.Errorf("rsvp-te: only IPv4 source addresses supported, got %s", localAddr)
		}
	}
	t := &rawTransport{sendFD: -1, pathFD: -1}
	if err := t.openSockets(localAddr); err != nil {
		t.closeSockets()
		return nil, errors.Join(err, t.closeErr)
	}
	t.recvCh = make(chan Packet, 64)
	t.closeC = make(chan struct{})
	t.done = make(chan struct{})
	go t.readLoop()
	return t, nil
}

func (t *rawTransport) openSockets(localAddr netip.Addr) error {
	fd, err := unix.Socket(unix.AF_INET, unix.SOCK_RAW|unix.SOCK_CLOEXEC|unix.SOCK_NONBLOCK, rsvpProtocol)
	if err != nil {
		return fmt.Errorf("rsvp-te: open receive socket needs CAP_NET_RAW: %w", err)
	}
	t.rx = os.NewFile(uintptr(fd), "rsvp-receive")
	t.rxRaw, err = t.rx.SyscallConn()
	if err != nil {
		return fmt.Errorf("rsvp-te: receive socket poller: %w", err)
	}
	// RFC 2205 Section 3.11.5: "Packets received for IP protocol 46 but not
	// addressed to the node must be diverted to the RSVP program for
	// processing, without being forwarded." IP_ROUTER_ALERT registers this
	// wildcard socket with Linux ip_call_ra_chain; binding to the router ID
	// would exclude ordinary local delivery to the other interface addresses.
	if err := unix.SetsockoptInt(fd, unix.IPPROTO_IP, unix.IP_ROUTER_ALERT, 1); err != nil {
		return fmt.Errorf("rsvp-te: receive transit Router Alert: %w", err)
	}
	if err := unix.SetsockoptInt(fd, unix.IPPROTO_IP, unix.IP_PKTINFO, 1); err != nil {
		return fmt.Errorf("rsvp-te: receive interface metadata: %w", err)
	}

	t.sendFD, err = unix.Socket(unix.AF_INET, unix.SOCK_RAW|unix.SOCK_CLOEXEC|unix.SOCK_NONBLOCK, rsvpProtocol)
	if err != nil {
		return fmt.Errorf("rsvp-te: open reply socket: %w", err)
	}
	if localAddr.Is4() {
		if err := unix.Bind(t.sendFD, &unix.SockaddrInet4{Addr: localAddr.As4()}); err != nil {
			return fmt.Errorf("rsvp-te: bind reply source %s: %w", localAddr, err)
		}
	}
	t.pathFD, err = unix.Socket(unix.AF_INET, unix.SOCK_RAW|unix.SOCK_CLOEXEC|unix.SOCK_NONBLOCK, unix.IPPROTO_RAW)
	if err != nil {
		return fmt.Errorf("rsvp-te: open PATH socket: %w", err)
	}
	if err := unix.SetsockoptInt(t.pathFD, unix.IPPROTO_IP, unix.IP_HDRINCL, 1); err != nil {
		return fmt.Errorf("rsvp-te: enable explicit PATH header: %w", err)
	}
	t.routes, err = netlink.NewHandle(unix.NETLINK_ROUTE)
	if err != nil {
		return fmt.Errorf("rsvp-te: open route lookup socket: %w", err)
	}
	return nil
}

func (t *rawTransport) Send(dst netip.Addr, msg []byte) error {
	if !dst.Is4() {
		return fmt.Errorf("rsvp-te: only IPv4 destinations supported, got %s", dst)
	}
	if len(msg) < rsvpHdrLen || len(msg) > maxRSVPPacket {
		return fmt.Errorf("rsvp-te: invalid message length %d", len(msg))
	}
	t.sendMu.Lock()
	defer t.sendMu.Unlock()
	if t.closed {
		return os.ErrClosed
	}
	routes, err := t.routes.RouteGet(dst.AsSlice())
	if err != nil {
		return fmt.Errorf("rsvp-te: resolve reply next hop %s: %w", dst, err)
	}
	if len(routes) != 1 || routes[0].LinkIndex <= 0 {
		return fmt.Errorf("rsvp-te: reply next hop %s does not select one interface", dst)
	}
	source, err := t.outgoingSource(&routes[0])
	if err != nil {
		return err
	}
	if len(msg) >= rsvpHdrLen {
		if err := unix.SetsockoptInt(t.sendFD, unix.IPPROTO_IP, unix.IP_TTL, int(msg[4])); err != nil {
			return fmt.Errorf("rsvp-te: set reply TTL: %w", err)
		}
	}
	// A RESV's IP source must name the adjacent interface, not a loopback
	// router ID: the upstream LSR uses it as the native MPLS next hop.
	oob := unix.PktInfo4(&unix.Inet4Pktinfo{Ifindex: int32(routes[0].LinkIndex), Spec_dst: source.As4()})
	var scratch [12]byte
	var buffers [4][]byte
	count := outgoingMessage(msg, source, scratch[:], buffers[:])
	n, err := unix.SendmsgBuffers(t.sendFD, buffers[:count], oob, &unix.SockaddrInet4{Addr: dst.As4()}, 0)
	if err != nil {
		return fmt.Errorf("rsvp-te: send reply to %s: %w", dst, err)
	}
	if n != len(msg) {
		return io.ErrShortWrite
	}
	return nil
}

// outgoingSource chooses an address actually assigned to the selected link.
// A route's preferred source alone is insufficient: it may name a loopback.
func (t *rawTransport) outgoingSource(route *netlink.Route) (netip.Addr, error) {
	link := &netlink.Device{LinkAttrs: netlink.LinkAttrs{Index: route.LinkIndex}}
	addresses, err := t.routes.AddrList(link, unix.AF_INET)
	if err != nil {
		return netip.Addr{}, fmt.Errorf("rsvp-te: interface %d addresses: %w", route.LinkIndex, err)
	}
	preferred, _ := netip.AddrFromSlice(route.Src)
	preferred = preferred.Unmap()
	var source netip.Addr
	for i := range addresses {
		if addresses[i].IPNet == nil {
			continue
		}
		address, ok := netip.AddrFromSlice(addresses[i].IPNet.IP)
		address = address.Unmap()
		if !ok || !address.Is4() || address.IsUnspecified() || address.IsMulticast() {
			continue
		}
		if address == preferred {
			return address, nil
		}
		if !source.IsValid() {
			source = address
		}
	}
	if !source.IsValid() {
		return netip.Addr{}, fmt.Errorf("rsvp-te: interface %d has no IPv4 source address", route.LinkIndex)
	}
	return source, nil
}

// outgoingMessage substitutes the selected interface in RSVP_HOP without
// mutating or copying the caller's payload. The four output slices borrow msg
// and twelve bytes of caller-provided scratch until sendmsg returns.
func outgoingMessage(msg []byte, source netip.Addr, scratch []byte, buffers [][]byte) int {
	buffers[0] = msg
	for off := rsvpHdrLen; off+objHdrLen <= len(msg); {
		length := int(binary.BigEndian.Uint16(msg[off : off+2]))
		if length < objHdrLen || length > len(msg)-off {
			return 1
		}
		if msg[off+2] == ClassRSVPHop && msg[off+3] == CTypeIPv4 && length == 12 {
			address := source.As4()
			if [4]byte(msg[off+4:off+8]) == address {
				return 1
			}
			copy(scratch[:rsvpHdrLen], msg[:rsvpHdrLen])
			copy(scratch[rsvpHdrLen:], address[:])
			checksum := binary.BigEndian.Uint16(msg[2:4])
			// RFC 2205 §3.1 permits zero to omit the checksum; preserve that.
			if checksum != 0 {
				// RFC 1624 equation 3, applied bytewise so even an odd object
				// offset retains the correct checksum-byte pairing.
				sum := uint32(^checksum)
				for i := range address {
					old, replacement := uint16(msg[off+4+i]), uint16(address[i])
					if (off+i)%2 == 0 {
						old <<= 8
						replacement <<= 8
					}
					sum += uint32(^old) + uint32(replacement)
				}
				for sum>>16 != 0 {
					sum = (sum & 0xffff) + (sum >> 16)
				}
				binary.BigEndian.PutUint16(scratch[2:4], ^uint16(sum))
			}
			buffers[0] = scratch[:rsvpHdrLen]
			buffers[1] = msg[rsvpHdrLen:off+4]
			buffers[2] = scratch[rsvpHdrLen : rsvpHdrLen+4]
			buffers[3] = msg[off+8:]
			return 4
		}
		off += length
	}
	return 1
}

func (t *rawTransport) SendPath(route PathRoute, msg []byte) error {
	if !route.Destination.Is4() || !route.Source.Is4() || !route.NextHop.Is4() ||
		(route.Lookup.IsValid() && !route.Lookup.Is4()) {
		return fmt.Errorf("rsvp-te: PATH requires IPv4 destination, source and next hop")
	}
	if route.Source.IsUnspecified() {
		return fmt.Errorf("rsvp-te: PATH source must be specified")
	}
	if len(msg) < rsvpHdrLen || len(msg) > maxRSVPPacket {
		return fmt.Errorf("rsvp-te: invalid PATH carriage length %d", len(msg))
	}
	t.sendMu.Lock()
	defer t.sendMu.Unlock()
	if t.closed {
		return os.ErrClosed
	}
	// A marked NextHop is the selected bypass's merge point. Its private
	// route supplies the MPLS stack and immediate gateway. The protected
	// SESSION destination can be beyond that merge point.
	lookup := route.Lookup
	if !lookup.IsValid() {
		lookup = route.NextHop
	}
	routes, err := t.routes.RouteGetWithOptions(lookup.AsSlice(), &netlink.RouteGetOptions{Mark: route.TableID, OifIndex: route.IfIndex})
	if err != nil {
		return fmt.Errorf("rsvp-te: resolve PATH next hop %s: %w", route.NextHop, err)
	}
	if len(routes) != 1 {
		return fmt.Errorf("rsvp-te: PATH next hop %s resolved to %d routes", route.NextHop, len(routes))
	}
	if route.TableID != 0 {
		if uint32(routes[0].Table) != route.TableID {
			return fmt.Errorf("rsvp-te: bypass table %d unavailable (lookup selected %d)", route.TableID, routes[0].Table)
		}
		encap, ok := routes[0].Encap.(*netlink.MPLSEncap)
		if !ok {
			return fmt.Errorf("rsvp-te: bypass table %d has no MPLS encapsulation", route.TableID)
		}
		if len(encap.Labels) == 0 {
			return fmt.Errorf("rsvp-te: bypass table %d has an empty MPLS stack", route.TableID)
		}
	}
	if routes[0].LinkIndex <= 0 {
		return fmt.Errorf("rsvp-te: PATH next hop %s has no output interface", route.NextHop)
	}
	sourceAddress, err := t.outgoingSource(&routes[0])
	if err != nil {
		return err
	}
	// The lock covers SO_MARK and sendmsg together. Replies use another socket
	// and cannot inherit the mark. fib-kernel's terminal mark rule prevents
	// fallback if a bypass disappears between this lookup and the send.
	if err := unix.SetsockoptInt(t.pathFD, unix.SOL_SOCKET, unix.SO_MARK, int(route.TableID)); err != nil {
		return fmt.Errorf("rsvp-te: select bypass table %d: %w", route.TableID, err)
	}

	// RFC 791 Section 3.1; RFC 2113 Section 2.1. Byte offsets:
	//   0        1       2..3       4..7       8    9    10..11
	// +--------+-----+------------+----------+----+----+----------+
	// | v4/IHL | TOS | total len  | ID/frag  |TTL | 46 | checksum |
	// +--------+-----+------------+----------+----+----+----------+
	// | source (12..15)           | destination (16..19)         |
	// +--------------------------+-----------------------------+
	// | Router Alert: 0x94, 4, 0, 0 (20..23)                     |
	// +--------------------------------------------------------+
	// Linux fills total length, identification and the IPv4 checksum.
	var header [24]byte
	header[0] = 0x46
	header[8] = msg[4] // RFC 2205 Section 3.1: Send_TTL is the outgoing IP TTL.
	header[9] = rsvpProtocol
	// RFC 2205 Section 3.1.3: "The IP source address of a Path message must
	// be an address of the sender it describes, while the destination address
	// must be the DestAddress for the session."
	source, destination := route.Source.As4(), route.Destination.As4()
	copy(header[12:16], source[:])
	copy(header[16:20], destination[:])
	// RFC 2205 Section 3.3: "Path, PathTear, and ResvConf messages must be
	// sent with the Router Alert IP option [RFC 2113] in their IP headers."
	header[20], header[21] = 0x94, 4
	// Leave ipi_spec_dst zero: an upstream sender's address need not be local.
	// The header owns that identity; IP_PKTINFO pins only the selected link.
	oob := unix.PktInfo4(&unix.Inet4Pktinfo{Ifindex: int32(routes[0].LinkIndex)})
	var scratch [12]byte
	var buffers [5][]byte
	buffers[0] = header[:]
	count := outgoingMessage(msg, sourceAddress, scratch[:], buffers[1:])
	n, err := unix.SendmsgBuffers(t.pathFD, buffers[:count+1], oob, &unix.SockaddrInet4{Addr: lookup.As4()}, 0)
	if err != nil {
		return fmt.Errorf("rsvp-te: send PATH via %s: %w", route.NextHop, err)
	}
	if n != len(header)+len(msg) {
		return io.ErrShortWrite
	}
	return nil
}

func (t *rawTransport) LocalAddresses() ([]InterfaceAddress, error) {
	t.sendMu.Lock()
	defer t.sendMu.Unlock()
	if t.closed {
		return nil, os.ErrClosed
	}
	assigned, err := t.routes.AddrList(nil, unix.AF_INET)
	if err != nil {
		return nil, fmt.Errorf("rsvp-te: local interface addresses: %w", err)
	}
	links, err := t.routes.LinkList()
	if err != nil {
		return nil, fmt.Errorf("rsvp-te: local interfaces: %w", err)
	}
	interfaces := make(map[int]*netlink.LinkAttrs, len(links))
	for _, link := range links {
		interfaces[link.Attrs().Index] = link.Attrs()
	}
	addresses := make([]InterfaceAddress, 0, len(assigned))
	for i := range assigned {
		if assigned[i].IPNet == nil {
			continue
		}
		address, ok := netip.AddrFromSlice(assigned[i].IPNet.IP)
		address = address.Unmap()
		if ok && address.Is4() && !address.IsUnspecified() && !address.IsMulticast() {
			link := interfaces[assigned[i].LinkIndex]
			if link != nil {
				addresses = append(addresses, InterfaceAddress{
					Address: address, IfIndex: link.Index, Up: link.Flags&net.FlagUp != 0,
					MTU: uint32(max(link.MTU, 0)),
				})
			}
		}
	}
	return addresses, nil
}

func (t *rawTransport) ResolveRoute(target netip.Prefix, destination netip.Addr, tableID uint32) (RouteInfo, error) {
	if !target.IsValid() || !target.Addr().Is4() {
		return RouteInfo{}, fmt.Errorf("rsvp-te: route target must be an IPv4 abstract node")
	}
	lookup := target.Addr()
	if target.Contains(destination) {
		lookup = destination
	}
	t.sendMu.Lock()
	defer t.sendMu.Unlock()
	if t.closed {
		return RouteInfo{}, os.ErrClosed
	}
	routes, err := t.routes.RouteGetWithOptions(lookup.AsSlice(), &netlink.RouteGetOptions{Mark: tableID})
	if err != nil {
		return RouteInfo{}, err
	}
	if len(routes) != 1 || routes[0].LinkIndex <= 0 || routes[0].Type != unix.RTN_UNICAST {
		return RouteInfo{}, fmt.Errorf("rsvp-te: no unicast route toward %s", target)
	}
	route := &routes[0]
	if tableID != 0 && uint32(route.Table) != tableID {
		return RouteInfo{}, fmt.Errorf("rsvp-te: forwarding context %d unavailable", tableID)
	}
	link, err := t.routes.LinkByIndex(route.LinkIndex)
	if err != nil {
		return RouteInfo{}, err
	}
	if link.Attrs().Flags&net.FlagUp == 0 || link.Attrs().MTU <= 0 {
		return RouteInfo{}, fmt.Errorf("rsvp-te: output interface %d is unavailable", route.LinkIndex)
	}
	mtu := uint32(link.Attrs().MTU)
	if route.MTU > 0 && uint32(route.MTU) < mtu {
		mtu = uint32(route.MTU)
	}
	next, _ := netip.AddrFromSlice(route.Gw)
	next = next.Unmap()
	if !next.IsValid() {
		if route.Via != nil {
			return RouteInfo{}, fmt.Errorf("rsvp-te: route toward %s has a non-IPv4 gateway", target)
		}
		if target.Bits() == 32 || target.Contains(destination) {
			next = lookup
		} else {
			// A connected prefix does not make its network address a router.
			// Select an actual known adjacent member, never invent a host.
			neighbors, err := t.routes.NeighList(route.LinkIndex, unix.AF_INET)
			if err != nil {
				return RouteInfo{}, err
			}
			for _, neighbor := range neighbors {
				address, ok := netip.AddrFromSlice(neighbor.IP)
				address = address.Unmap()
				if ok && target.Contains(address) && neighbor.State&(netlink.NUD_INCOMPLETE|netlink.NUD_FAILED) == 0 &&
					(!next.IsValid() || address.Less(next)) {
					next = address
				}
			}
			if !next.IsValid() {
				return RouteInfo{}, fmt.Errorf("rsvp-te: no adjacent member known in %s", target)
			}
			lookup = next
		}
	}
	if !next.Is4() || next.IsUnspecified() || next.IsMulticast() {
		return RouteInfo{}, fmt.Errorf("rsvp-te: route toward %s has no usable IPv4 next hop", target)
	}
	return RouteInfo{NextHop: next, IfIndex: route.LinkIndex, Lookup: lookup, MTU: mtu}, nil
}

func (t *rawTransport) Recv() <-chan Packet { return t.recvCh }

// Close MUST be called by the owner. It wakes the poller, stops the receive
// worker, and waits until Recv is closed. Repeated calls return the same error.
func (t *rawTransport) Close() error {
	t.once.Do(func() {
		t.sendMu.Lock()
		t.closed = true
		close(t.closeC)
		t.closeSockets()
		t.sendMu.Unlock()
		<-t.done
	})
	return t.closeErr
}

func (t *rawTransport) closeSockets() {
	if t.rx != nil {
		t.closeErr = errors.Join(t.closeErr, t.rx.Close())
	}
	if t.sendFD >= 0 {
		t.closeErr = errors.Join(t.closeErr, unix.Close(t.sendFD))
	}
	if t.pathFD >= 0 {
		t.closeErr = errors.Join(t.closeErr, unix.Close(t.pathFD))
	}
	if t.routes != nil {
		t.routes.Close()
	}
}

func (t *rawTransport) readLoop() {
	defer close(t.done)
	defer close(t.recvCh)
	buf := make([]byte, 65535)
	oob := make([]byte, unix.CmsgSpace(unix.SizeofInet4Pktinfo))
	// The receive worker lasts until Close or a permanent socket error. RawConn
	// polls the nonblocking descriptor, so closing rx wakes an idle receive.
	for {
		var n, oobn, flags int
		var recvErr error
		err := t.rxRaw.Read(func(fd uintptr) bool {
			n, oobn, flags, _, recvErr = unix.Recvmsg(int(fd), buf, oob, 0)
			return recvErr != unix.EAGAIN && recvErr != unix.EINTR
		})
		if err != nil {
			if !errors.Is(err, os.ErrClosed) {
				logger().Warn("rsvp-te: receive poller stopped", "err", err)
			}
			return
		}
		if recvErr != nil {
			logger().Warn("rsvp-te: receive socket stopped", "err", recvErr)
			return
		}
		// recvmsg consumes even a truncated datagram; never decode its prefix.
		if flags&(unix.MSG_TRUNC|unix.MSG_CTRUNC) != 0 {
			continue
		}
		packet, ok := receivedIPv4Packet(buf[:n], oob[:oobn])
		if !ok {
			continue
		}
		// The channel consumer owns this copy after the receive buffer is reused.
		payload := make([]byte, len(packet.Payload))
		copy(payload, packet.Payload)
		packet.Payload = payload
		select {
		case t.recvCh <- packet:
		case <-t.closeC:
			return
		}
	}
}

func receivedIPv4Packet(data, oob []byte) (Packet, bool) {
	if len(data) < 20 {
		return Packet{}, false
	}
	if data[0]>>4 != 4 || data[9] != rsvpProtocol {
		return Packet{}, false
	}
	ihl := int(data[0]&0x0f) * 4
	total := int(binary.BigEndian.Uint16(data[2:4]))
	if ihl < 20 || ihl > total || total > len(data) {
		return Packet{}, false
	}
	if total-ihl > maxRSVPPacket {
		return Packet{}, false
	}
	// RFC 2205 Section 3.11.5: "On a router or multi-homed host, the identity
	// of the interface (real or virtual) on which a diverted message is
	// received, as well as the IP source address and IP TTL with which it
	// arrived, must also be available to the RSVP process."
	packet := Packet{
		Src: netip.AddrFrom4([4]byte(data[12:16])),
		Dst: netip.AddrFrom4([4]byte(data[16:20])),
		TTL: data[8],
		Payload: data[ihl:total],
	}
	// Each iteration consumes one aligned control message from the bounded
	// ancillary buffer. Other socket metadata does not replace IP_PKTINFO.
	for len(oob) >= unix.CmsgLen(0) {
		hdr, body, rest, err := unix.ParseOneSocketControlMessage(oob)
		if err != nil {
			return Packet{}, false
		}
		oob = rest
		if hdr.Level != unix.IPPROTO_IP || hdr.Type != unix.IP_PKTINFO {
			continue
		}
		if len(body) < unix.SizeofInet4Pktinfo {
			return Packet{}, false
		}
		packet.IfIndex = int(int32(binary.NativeEndian.Uint32(body[:4])))
	}
	return packet, packet.IfIndex > 0
}
