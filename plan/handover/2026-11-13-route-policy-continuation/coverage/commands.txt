# For host jobs copy ONLY actual JSON lines from the native job logfile to a
# sibling *.tests.jsonl. This selects top-level results and prints EVERY name,
# pass/fail/skip counts, then refuses empty population, failed or skipped tests.
# Test names containing / are subtests and may be listed separately if wanted.
# Reject a (cached) line before attributing these events to fresh execution;
# rerun only an actually cached invocation with -count=1 if fresh proof is owed.
rg '\(cached\)' ACTUAL.tests.jsonl
jq -s 'map(select(.Test != null and (.Test|contains("/")|not) and (.Action=="pass" or .Action=="fail" or .Action=="skip"))) | {named:map({package:.Package,test:.Test,outcome:.Action}),pass:map(select(.Action=="pass"))|length,fail:map(select(.Action=="fail"))|length,skip:map(select(.Action=="skip"))|length}' ACTUAL.tests.jsonl
jq -se 'map(select(.Test != null and (.Test|contains("/")|not) and (.Action=="pass" or .Action=="fail" or .Action=="skip"))) as $r | ($r|length)>0 and all($r[]; .Action=="pass")' ACTUAL.tests.jsonl
