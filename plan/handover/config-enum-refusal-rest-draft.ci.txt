# Saved draft, not a test yet. Restore to test/draft/plugin/api-config-enum-refusal.ci.
# It passed once alone and timed out under the 80-run load proof: the REST
# listener port 18088 is fixed, so parallel copies collide. Give it a leased
# port (as the peer uses $PORT) before proving and promoting it. Its fixture
# scenario `plugin/api-config-enum-refusal` is committed in
# internal/test/fixture/plugin_fixture_01.go and plugin_fixture_01_api.go.
#
# Test: a REST config session refuses a list key at set and a value at commit
# VALIDATES: AC-8, AC-43 (spec-config-enum-refusal-proposes-valid-values) --
# a session set of `ospf redistribute bogus nssa-propagate true` answers HTTP
# 400 with the refusal and `fix: set ospf redistribute <connected|...>`; a
# session holding `route-refresh requre` is refused at commit with HTTP 400,
# the refusal's problem and accepted values, no `line N:`, and
# `fix: set bgp peer peer1 session capability route-refresh require` last;
# the active config is unchanged.
# PREVENTS: the REST session answering the file-form fix, a bare error, or a
# code other than 400 for a refused value or key.
# Why not a unit test: the hook test (TestConfigValidationHookEnumRefusal)
# proves the text at the hook; only the daemon proves the REST listener, the
# session editor and the hook are wired so a client receives it.

option=storage:value=disk
option=timeout:value=30s

stdin=peer:terminator=EOF_PEER
option=tcp_connections:value=1
expect=bgp:conn=1:seq=1:hex=FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFF00170200000000
EOF_PEER


tmpfs=api-config-enum-refusal.conf:terminator=EOF_CONF
plugin {
    external api-enum-refusal {
        run "le test fixture plugin/api-config-enum-refusal"
        encoder json
    }
}
environment {
    api-server {
        token "secret";
        rest {
            enabled true;
            server main {
                ip 127.0.0.1;
                port 18088;
            }
        }
    }
}

bgp {
    router-id 1.2.3.4
    peer peer1 {
        connection {
            remote {
                ip 127.0.0.1
            }
            local {
                ip 127.0.0.1
                accept false
            }
        }
        session {
            asn {
                local 1
                remote 1
            }
            router-id 1.2.3.4
            family {
                ipv4/unicast { prefix { maximum 10000; } }
            }
            capability {
                graceful-restart disable
            }
        }
        behavior {
            group-updates disable
        }
    }
}
EOF_CONF

cmd=background:seq=1:exec=le test peer --port $PORT:stdin=peer
cmd=foreground:seq=2:exec=ze start api-config-enum-refusal.conf:timeout=30s
expect=exit:code=0
expect=stderr:contains=OK: REST session set refused the list key bogus
expect=stderr:contains=OK: REST session commit refused route-refresh requre
expect=key:path=meta/config/api-config-enum-refusal.conf/active:exists=true
expect=key:path=meta/config/api-config-enum-refusal.conf/candidate:absent=true
expect=key:glob=object/*:count=1
expect=key:glob=object/*:not-contains=requre
expect=key:glob=object/*:not-contains=redistribute
